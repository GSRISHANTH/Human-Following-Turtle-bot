#   mediapipe 1.1.1 | opencv-python 5.0.0.93 | numpy 2.5.3 | Flask 3.1.3

# Ranges, not single pins, because no one numpy release covers 3.10 - 3.14 (numpy >= 2.3 dropped 3.10).

# opencv-python, keep it at exactly the same version, or uninstall it (pip uninstall opencv-python).
opencv-contrib-python>=4.9,<6

numpy>=1.26,<2.3; python_version < "3.11"
numpy>=2.0,<3; python_version >= "3.11"

# Web dashboard (port 8000) and Server-Sent Events telemetry.
